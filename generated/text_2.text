async function sendContact(
  p: Record<string, unknown>,
  proxyUrl: string,
  token: string,
) {
  const data = await callProxy(proxyUrl, token, {
    provider: "resend",
    endpoint: "/emails",
    payload: {
      from: "HolaSki <noreply@holaski.mx>",
      to: ["hola@holaski.mx"],
      subject: `Nuevo mensaje de ${p.name}`,
      html: `<p><strong>${p.name}</strong> (${p.email})</p><p>${p.message}</p>`,
    },
  });
  return { sent: true, id: data?.id };
}
